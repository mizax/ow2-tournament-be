rule("UltSaver: On player join recover")
{
	event
	{
		Player Joined Match;
		All;
		Все;
	}

	conditions
	{
		Is Waiting For Players == False;
		Is Between Rounds == False;
	}

	actions
	{
		Event Player.key = Event Player.Meta_playerName;
		Event Player.key_0 = Event Player.Meta_playerName;
		If(Team Of(Event Player) == Team 1 ? Global._ultSaverTeam1Hero_Values[Index Of Array Value(Global._ultSaverTeam1Hero_Keys,
			Event Player.key)] : Global._ultSaverTeam2Hero_Values[Index Of Array Value(Global._ultSaverTeam2Hero_Keys, Event Player.key_0)
			]);
			Event Player.Ult_isPlayerReconnecting = True;
			Event Player.key_1 = Event Player.Meta_playerName;
			Event Player.key_2 = Event Player.Meta_playerName;
			Event Player.Ult_reconnectingPlayerChargePerc = Team Of(Event Player)
				== Team 1 ? Global._ultSaverTeam1Charge_Values[Index Of Array Value(Global._ultSaverTeam1Charge_Keys, Event Player.key_1)
				] : Global._ultSaverTeam2Charge_Values[Index Of Array Value(Global._ultSaverTeam2Charge_Keys, Event Player.key_2)];
			Event Player.key_3 = Event Player.Meta_playerName;
			Event Player.key_4 = Event Player.Meta_playerName;
			Start Forcing Player To Be Hero(Event Player, Team Of(Event Player)
				== Team 1 ? Global._ultSaverTeam1Hero_Values[Index Of Array Value(Global._ultSaverTeam1Hero_Keys, Event Player.key_3)
				] : Global._ultSaverTeam2Hero_Values[Index Of Array Value(Global._ultSaverTeam2Hero_Keys, Event Player.key_4)]);
		End;
	}
}

		Event Player.Ult_reconnectingPlayerChargePerc = 0;
	}
}

rule("UltSaver: Track player's current hero")
{
	event
	{
		Ongoing - Each Player;
		All;
		Все;
	}

	conditions
	{
		Is Waiting For Players == False;
		(Team Of(Event Player) == Team 1 ? Global._ultSaverTeam1Hero_Values[Index Of Array Value(Global._ultSaverTeam1Hero_Keys,
			Event Player.key_5)] != Hero Of(Event Player) : Global._ultSaverTeam2Hero_Values[Index Of Array Value(
			Global._ultSaverTeam2Hero_Keys, Event Player.key_6)] != Hero Of(Event Player)) == True;
	}

	actions
	{
		Event Player.key_5 = Event Player.Meta_playerName;
		Event Player.key_6 = Event Player.Meta_playerName;
		If(Team Of(Event Player) == Team 1);
			Event Player.index = Index Of Array Value(Global._ultSaverTeam1Hero_Keys, Event Player.Meta_playerName);
			If(Event Player.index == -1);
				Modify Global Variable(_ultSaverTeam1Hero_Keys, Append To Array, Event Player.Meta_playerName);
				Modify Global Variable(_ultSaverTeam1Hero_Values, Append To Array, Hero Of(Event Player));
			Else;
				Global._ultSaverTeam1Hero_Values[Event Player.index] = Hero Of(Event Player);
			End;
		Else;
			Event Player.index_0 = Index Of Array Value(Global._ultSaverTeam2Hero_Keys, Event Player.Meta_playerName);
			If(Event Player.index_0 == -1);
				Modify Global Variable(_ultSaverTeam2Hero_Keys, Append To Array, Event Player.Meta_playerName);
				Modify Global Variable(_ultSaverTeam2Hero_Values, Append To Array, Hero Of(Event Player));
			Else;
				Global._ultSaverTeam2Hero_Values[Event Player.index_0] = Hero Of(Event Player);
			End;
		End;
		Wait(1, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("UltSaver: Track player's ultimate charge")
{
	event
	{
		Ongoing - Each Player;
		All;
		Все;
	}

	conditions
	{
		Is Waiting For Players == False;
		Is Duplicating(Event Player) == False;
		(Team Of(Event Player) == Team 1 ? Global._ultSaverTeam1Charge_Values[Index Of Array Value(Global._ultSaverTeam1Charge_Keys,
			Event Player.key_7)] != Ultimate Charge Percent(Event Player) : Global._ultSaverTeam2Charge_Values[Index Of Array Value(
			Global._ultSaverTeam2Charge_Keys, Event Player.key_8)] != Ultimate Charge Percent(Event Player)) == True;
	}

	actions
	{
		Event Player.key_7 = Event Player.Meta_playerName;
		Event Player.key_8 = Event Player.Meta_playerName;
		Abort If(Is In Alternate Form(Event Player) == True && Hero Of(Event Player) == Hero(D.Va));
		If(Team Of(Event Player) == Team 1);
			Event Player.index_1 = Index Of Array Value(Global._ultSaverTeam1Charge_Keys, Event Player.Meta_playerName);
			If(Event Player.index_1 == -1);
				Modify Global Variable(_ultSaverTeam1Charge_Keys, Append To Array, Event Player.Meta_playerName);
				Modify Global Variable(_ultSaverTeam1Charge_Values, Append To Array, Ultimate Charge Percent(Event Player));
			Else;
				Global._ultSaverTeam1Charge_Values[Event Player.index_1] = Ultimate Charge Percent(Event Player);
			End;
		Else;
			Event Player.index_2 = Index Of Array Value(Global._ultSaverTeam2Charge_Keys, Event Player.Meta_playerName);
			If(Event Player.index_2 == -1);
				Modify Global Variable(_ultSaverTeam2Charge_Keys, Append To Array, Event Player.Meta_playerName);
				Modify Global Variable(_ultSaverTeam2Charge_Values, Append To Array, Ultimate Charge Percent(Event Player));
			Else;
				Global._ultSaverTeam2Charge_Values[Event Player.index_2] = Ultimate Charge Percent(Event Player);
			End;
		End;
		Wait(1, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("Logs: Hero Spawn/Swap")
{
	event
	{
		Ongoing - Each Player;
		All;
		Все;
	}

	conditions
	{
		Global.Logs_EnableLogTracker == True;
		(Global.Logs_TrackHeroSwaps || Global.Logs_PlayerSummaryCount != 2) == True;
		Hero Of(Event Player) != Event Player.Logs_LastHero;
		Is Match Complete == False;
	}

	actions
	{
		If(Global.Logs_TrackHeroSwaps && Array Contains(Все герои, Hero Of(Event Player)));
			If(Array Contains(Все герои, Event Player.Logs_LastHero));
				Log to Inspector(Custom String(",hero_swap,{0},{1},{2}", Global.Scrim_MatchTime, Custom String("{0},{1}", Team Of(Event Player),
					Event Player), Custom String("{0},{1},{2}", Hero Of(Event Player), Event Player.Logs_LastHero,
					Global.Scrim_MatchTime - Event Player.Logs_LastHeroTime)));
			Else;
				Log to Inspector(Custom String(",hero_spawn,{0},{1},{2}", Global.Scrim_MatchTime, Custom String("{0},{1}", Team Of(Event Player),
					Event Player), Custom String("{0},{1},{2}", Hero Of(Event Player), Event Player.Logs_LastHero,
					Global.Scrim_MatchTime - Event Player.Logs_LastHeroTime)));
			End;
		End;
		If(!Array Contains(Event Player.Logs_HeroesPlayedArray, Hero Of(Event Player)) && Array Contains(Все герои, Hero Of(
			Event Player)));
			Modify Player Variable(Event Player, Logs_HeroesPlayedArray, Append To Array, Hero Of(Event Player));
		End;
		If(Array Contains(Все герои, Hero Of(Event Player)) && Array Contains(Все герои, Event Player.Logs_LastHero));
			Call Subroutine(CalcHeroPlayedTime);
		End;
		Event Player.Logs_LastHero = Hero Of(Event Player);
		Event Player.Logs_LastHeroTime = Global.Scrim_MatchTime;
	}
}
